# Scripted effects. Scope noted on each.
# Ark colonies are colony scope (carried by a ship); vanilla iterates them with
# *_owned_colony + is_planet_class = pc_ark (events/nomads_arkship_events.txt:5005-5008).

# colony scope (an Ark colony). Gives the Ark its Tree of Life.
nw_plant_heartwood_core = {
	remove_modifier = nw_withered_heartwood
	if = {
		limit = { NOT = { has_deposit = d_nw_heartwood_core } }
		add_deposit = d_nw_heartwood_core
	}
}

# colony scope. The tree dies back.
nw_wither_heartwood = {
	remove_deposit = d_nw_heartwood_core
	add_modifier = { modifier = nw_withered_heartwood }
	owner = {
		if = {
			limit = { NOT = { has_country_flag = nw_first_wither_seen } }
			set_country_flag = nw_first_wither_seen
			country_event = { id = nw_uprooted.20 }
		}
	}
}

# colony scope. Hull threshold depends on class: 25%, or 12.5% for the Thornwood (Living Hull, decision D11).
# has_hp_percentage is a ship-scope 0-1 fraction (events/cosmic_storms_events_1.txt:822);
# ship? reaches a colony's carrier ship (events/game_start.txt:1792).
nw_check_wither = {
	if = {
		limit = {
			nw_ark_has_heartwood = yes
			OR = {
				has_planet_flag = nw_debug_force_wither	# test hook
				AND = {
					NOT = { has_planet_flag = nw_ark_thornwood }
					ship? = { has_hp_percentage < 0.25 }
				}
				AND = {
					has_planet_flag = nw_ark_thornwood
					ship? = { has_hp_percentage < 0.125 }
				}
			}
		}
		nw_wither_heartwood = yes
	}
}

# colony scope. Marks a freshly spawned or upgraded Ark colony with class and tier flags.
# Parameters: CLASS = heartwood|lumen|thornwood, TIER = 1|2|3
nw_tag_ark = {
	set_planet_flag = nw_tree_ark
	remove_planet_flag = nw_ark_tier_1
	remove_planet_flag = nw_ark_tier_2
	remove_planet_flag = nw_ark_tier_3
	set_planet_flag = nw_ark_tier_$TIER$
	remove_planet_flag = nw_ark_heartwood
	remove_planet_flag = nw_ark_lumen
	remove_planet_flag = nw_ark_thornwood
	set_planet_flag = nw_ark_$CLASS$
}

# country scope. Gives the starting Ark the class the player picked in the empire designer
# (decision D3-B; called by nw_uprooted.2, which reads the starting_<class>_arkship flag).
# Parameter: CLASS = heartwood|lumen|thornwood.
# For nomads, capital_scope is the capital Ark colony (vanilla uses it so in
# common/scripted_effects/01_start_of_game_effects.txt:5000-5002).
nw_apply_starting_ark_class = {
	capital_scope = {
		if = {
			limit = { nw_is_tree_ark_colony = yes }
			nw_tag_ark = { CLASS = $CLASS$ TIER = 1 }
			nw_plant_heartwood_core = yes
			remove_modifier = nw_heartwood_trait
			remove_modifier = nw_lumen_trait
			remove_modifier = nw_thornwood_trait
			add_modifier = { modifier = nw_$CLASS$_trait }
		}
	}
}

# Identity triggers for the mod. Everything else in the mod gates on these.

# The flag branch keeps the mod's content on after Take Root, when the empire's origin becomes
# origin_wilderness (decision D7-W); nw_uprooted.100 sets the flag explicitly (plan P19).
nw_is_uprooted_empire = {
	OR = {
		has_origin = origin_nw_uprooted
		has_country_flag = nw_uprooted
	}
}

# Colony scope. True for a colony that lives on one of our Tree Arks.
# Vanilla tests an Arkship colony with is_planet_class = pc_ark
# (common/planet_classes/06_planet_classes_nomads.txt:1; e.g. inline_scripts/megastructures/arkship.txt:69).
nw_is_tree_ark_colony = {
	exists = owner
	owner = { nw_is_uprooted_empire = yes }
	is_planet_class = pc_ark
}

nw_ark_has_heartwood = {
	has_deposit = d_nw_heartwood_core
}

nw_ark_is_withered = {
	has_modifier = nw_withered_heartwood
}

nw_ark_is_elder = {
	has_planet_flag = nw_ark_tier_3
}

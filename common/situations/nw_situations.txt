# Nurture situation: grows an Ark one tier. Started by the Nurture decision/button (docs/02 §3.3).
# Syntax per common/situations/99_README_SITUATIONS.txt. Vanilla Ark-targeted template:
# situation_grand_refurbishment (common/situations/14_nomads_situations.txt:1225-1448), started with
# start_situation = { type = ... target = <ark colony> } (events/nomads_arkship_events.txt:8375-8385).
# Scopes: this/root = situation, owner = country, target = the Ark colony (README :22).
# Decisions D4 (food upkeep) and D5 (Nurture replaces the vanilla upgrade) apply.

situation_nw_nurture_ark = {
	picture = GFX_evt_nw_seed
	category = positive
	start_value = 0
	stages = {
		nw_nurture_stage_1 = { icon = GFX_situation_stage_1 icon_background = GFX_situation_stage_frame_green end = 50 }
		nw_nurture_stage_2 = { icon = GFX_situation_stage_2 icon_background = GFX_situation_stage_frame_green end = 100 }
	}
	on_start = {
		set_situation_flag = nw_nurture_started
	}
	on_progress_complete = {
		target = {
			custom_tooltip = nw_nurture_complete_tt
			owner = { country_event = { id = nw_uprooted.50 } }
		}
		# <<VANILLA: the tier-up is new logic (decision D5-A, planned in plan-missing-scripts.md P9). Vanilla sets an
		#   Ark's tier from the colony with fleet = { starbase? = { set_starbase_size = <class>_arkship_tier_N } }
		#   (events/game_start.txt:3487-3503).>>
		destroy_situation = this	# required by README :55-57
	}
	on_abort = { }
	monthly_progress = {
		base = 1
		modifier = {
			factor = 1.5
			desc = nw_docked_at_roothold
			# Ark shares a system with one of our Waystations
			# (any_fleet_in_system + is_waystation_ship: megastructures/30_nomad_waystation.txt:155-157).
			target = {
				solar_system? = {
					any_fleet_in_system = {
						is_waystation_ship = yes
						owner = { is_same_value = root.owner }
					}
				}
			}
		}
		modifier = {
			factor = 2
			desc = nw_near_living_world
			target = {
				solar_system? = {
					any_system_planet = {
						OR = {
							is_planet_class = pc_gaia
							has_deposit = d_nw_heartwood_core
						}
					}
				}
			}
		}
		modifier = {
			factor = 0
			desc = nw_food_starved
			# D4: Nurture's recurring cost is food, so growth pauses when the food stockpile runs dry.
			# Country stockpile test syntax: common/scripted_actions/03_arkships.txt:648-652.
			owner = { resource_stockpile_compare = { resource = food value <= 0 } }
		}
		# Approach effects. Situations have no per-approach monthly_progress; vanilla keys them on
		# current_situation_approach here (common/situations/14_nomads_situations.txt:1250-1272).
		modifier = {
			factor = 2
			desc = nw_nurture_forced_bloom
			current_situation_approach = nw_nurture_forced_bloom
		}
		modifier = {
			factor = 0
			desc = nw_nurture_deep_dormancy
			current_situation_approach = nw_nurture_deep_dormancy
		}
	}
	# D4: the monthly drain is food upkeep on the approach (resources syntax: README :188-193;
	# vanilla example common/situations/14_nomads_situations.txt:1281-1285). Design values, balance pending.
	approach = {
		name = nw_nurture_slow_growth
		icon = GFX_situation_approach_heart
		icon_background = GFX_situation_approach_bg_green
		default = yes
		resources = {
			category = situations
			upkeep = { food = 15 }
		}
	}
	approach = {
		name = nw_nurture_forced_bloom
		icon = GFX_situation_approach_hungry
		icon_background = GFX_situation_approach_bg_red
		# Applies only while picked (README :186). Replaces the on_select/on_deselect pair; on_deselect does not exist.
		target_modifier = { ship_hull_mult = -0.25 }
		resources = {
			category = situations
			upkeep = { food = 45 }
		}
	}
	approach = {
		name = nw_nurture_deep_dormancy
		icon = GFX_situation_approach_shrug
		icon_background = GFX_situation_approach_bg_yellow
		target_modifier = { ship_hull_mult = 0.20 }
	}
}

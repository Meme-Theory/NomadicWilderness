country_event = {
	id = nw_uprooted.1
	title = "nw_uprooted.1.name"
	desc = "nw_uprooted.1.desc"
	picture = GFX_evt_nw_uprooting
	show_sound = event_default
	is_triggered_only = yes

	trigger = {
		nw_is_uprooted_empire = yes
		NOT = { has_country_flag = nw_opening_done }
	}

	immediate = {
		set_country_flag = nw_opening_done
	}

	option = {
		name = "nw_uprooted.1.a"   # The star was dying.
		set_country_flag = nw_blight_star
		country_event = { id = nw_uprooted.2 }
	}
	option = {
		name = "nw_uprooted.1.b"   # Something came.
		set_country_flag = nw_blight_visitor
		country_event = { id = nw_uprooted.2 }
	}
	option = {
		name = "nw_uprooted.1.c"   # We ate our own world.
		set_country_flag = nw_blight_overgrowth
		country_event = { id = nw_uprooted.2 }
	}
}

# Memory of the Grove: an Ark returns to the Rootscar after year 20.
# <<VANILLA: hook to the fleet-enters-system on_action, filtered on nw_rootscar_system flag>>
country_event = {
	id = nw_uprooted.30
	title = "nw_uprooted.30.name"
	desc = "nw_uprooted.30.desc"
	picture = GFX_evt_nw_rootscar
	show_sound = event_default
	is_triggered_only = yes
	trigger = {
		nw_is_uprooted_empire = yes
		years_passed > 20
		NOT = { has_country_flag = nw_memory_done }
	}
	immediate = { set_country_flag = nw_memory_done }
	option = {
		name = "nw_uprooted.30.a"   # Take the dormant sapling.
		custom_tooltip = nw_uprooted.30.a.tt
		hidden_effect = {
			event_target:nw_rootscar_planet = { remove_deposit = d_nw_dormant_sapling }
			set_country_flag = nw_has_emergency_sapling
		}
	}
	option = {
		name = "nw_uprooted.30.b"   # Leave it. Remember.
		add_modifier = { modifier = nw_memory_of_the_grove }
	}
}
